CPE Segment Tree Range Minimum Query

難度：中等
來源：原創（線段樹 (Segment Tree)：區間最小值查詢與單點更新）

== 題目 ==
給定 n 個整數，支援兩種操作："U pos val" 將索引 pos (0-indexed) 的值改為 val；"Q l r" 查詢區間 [l,r] 的最小值。請使用線段樹實作。

輸入說明：每組第一行 n，第二行 n 個整數；接著一行操作數 q，再 q 行操作。n=0 表示輸入結束。
輸出說明：依序輸出每個 Q 操作的結果。

== 參考解法 (Python) ==
import sys

class SegTree:
    def __init__(self, arr):
        self.n = len(arr)
        self.tree = [0] * (4 * self.n)
        if self.n:
            self.build(arr, 1, 0, self.n - 1)

    def build(self, arr, node, l, r):
        if l == r:
            self.tree[node] = arr[l]
            return
        mid = (l + r) // 2
        self.build(arr, node * 2, l, mid)
        self.build(arr, node * 2 + 1, mid + 1, r)
        self.tree[node] = min(self.tree[node * 2], self.tree[node * 2 + 1])

    def update(self, node, l, r, pos, val):
        if l == r:
            self.tree[node] = val
            return
        mid = (l + r) // 2
        if pos <= mid:
            self.update(node * 2, l, mid, pos, val)
        else:
            self.update(node * 2 + 1, mid + 1, r, pos, val)
        self.tree[node] = min(self.tree[node * 2], self.tree[node * 2 + 1])

    def query(self, node, l, r, ql, qr):
        if qr < l or r < ql:
            return float('inf')
        if ql <= l and r <= qr:
            return self.tree[node]
        mid = (l + r) // 2
        return min(self.query(node * 2, l, mid, ql, qr),
                    self.query(node * 2 + 1, mid + 1, r, ql, qr))

def main():
    data = sys.stdin.read().split('\n')
    idx = 0
    out = []
    while idx < len(data):
        line = data[idx].strip(); idx += 1
        if line == '':
            continue
        n = int(line)
        if n == 0:
            break
        arr = list(map(int, data[idx].split())); idx += 1
        st = SegTree(arr)
        q = int(data[idx].strip()); idx += 1
        for _ in range(q):
            parts = data[idx].split(); idx += 1
            if parts[0] == 'U':
                pos, val = int(parts[1]), int(parts[2])
                st.update(1, 0, n - 1, pos, val)
            else:
                l, r = int(parts[1]), int(parts[2])
                out.append(str(st.query(1, 0, n - 1, l, r)))
    print('\n'.join(out))

main()
