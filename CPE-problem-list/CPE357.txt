CPE 11624 Fire!

難度：困難
來源：改編自 UVa Online Judge #11624 (CPE 大學程式能力檢定經典題)

== 題目 ==
Joe 位於迷宮中 (座標 J)，迷宮某些格子已經著火 (F)，火每分鐘會朝上下左右四個方向蔓延一格；Joe 每分鐘也可以往上下左右移動一格 (不能進入牆壁 #，也不能進入當下已經著火的格子)。Joe 可以從迷宮邊界的任一格「離開」迷宮 (只要那格不是牆)。求 Joe 能安全逃出迷宮的最短時間；若不可能則輸出 IMPOSSIBLE。

輸入說明：第一行為測試個數。每組第一行 R C，接著 R 行each C 個字元 (#, ., J, F，恰有一個 J)。
輸出說明：每組輸出最短逃出時間，或 IMPOSSIBLE。

== 參考解法 (Python) ==
import sys
from collections import deque

def main():
    data = sys.stdin.read().split('\n')
    idx = 0
    t = int(data[idx].strip()); idx += 1
    out = []
    for _ in range(t):
        while data[idx].strip() == '':
            idx += 1
        r, c = map(int, data[idx].split()); idx += 1
        grid = [data[idx + i] for i in range(r)]
        idx += r
        fire_time = [[-1] * c for _ in range(r)]
        joe_time = [[-1] * c for _ in range(r)]
        fq = deque()
        jstart = None
        for i in range(r):
            for j in range(c):
                if grid[i][j] == 'F':
                    fire_time[i][j] = 0
                    fq.append((i, j))
                elif grid[i][j] == 'J':
                    jstart = (i, j)
        while fq:
            x, y = fq.popleft()
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nx, ny = x + dx, y + dy
                if 0 <= nx < r and 0 <= ny < c and grid[nx][ny] != '#' and fire_time[nx][ny] == -1:
                    fire_time[nx][ny] = fire_time[x][y] + 1
                    fq.append((nx, ny))
        jq = deque([jstart])
        joe_time[jstart[0]][jstart[1]] = 0
        answer = -1
        while jq:
            x, y = jq.popleft()
            if x == 0 or x == r - 1 or y == 0 or y == c - 1:
                answer = joe_time[x][y] + 1
                break
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nx, ny = x + dx, y + dy
                if 0 <= nx < r and 0 <= ny < c and grid[nx][ny] != '#' and joe_time[nx][ny] == -1:
                    nt = joe_time[x][y] + 1
                    if fire_time[nx][ny] == -1 or nt < fire_time[nx][ny]:
                        joe_time[nx][ny] = nt
                        jq.append((nx, ny))
        out.append(str(answer) if answer != -1 else 'IMPOSSIBLE')
    print('\n'.join(out))

main()
